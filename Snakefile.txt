"""Snakemake file that runs analysis."""


import pandas as pd


pacbio_runs = pd.read_csv("data/PacBio_runs.csv").set_index("run")["fastq"].to_dict()


rule all:
    input:
        expand("results/alignments/{run}.paf", run=pacbio_runs),


rule align:
    """Align FASTQs with minimap2."""
    input:
        ref="data/spike.fasta",
        fastq=lambda wc: pacbio_runs[wc.run],
    output:
        paf="results/alignments/{run}.paf",
    conda:
        "environment.yml"
    threads: 3
    shell:
        "minimap2 -t {threads} --cs {input.ref} {input.fastq} > {output.paf}"
